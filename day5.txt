4. Maximum and Minimum

7.Palindrome
n = int(input())

if n < 0:
    print("false")
else:
    original = n
    reverse = 0

    while n != 0:
        digit = n % 10
        reverse = reverse * 10 + digit
        n = n // 10

    if original == reverse:
        print("true")
    else:
        print("false")

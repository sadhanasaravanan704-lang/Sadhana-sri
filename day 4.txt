5.Smart Shop
cost1,quantity1= map(int,input().split())
cost2,quantity2=map(int,input().split())
cost3,quantity3=map(int,input().split())
budget=int(input())
total_before_tax = (cost1*quantity1)+(cost2*quantity2)+(cost3*quantity3)
tax = total_before_tax*0.10
total_after_tax = total_before_tax+tax
print(f"Total (Before Tax): {total_before_tax:.2f}")
print(f"Total (After Tax): {total_after_tax:.2f}")

if total_after_tax <= budget:
    print("Within Budget")
else:
    print("Over Budget")

7. Triangle Type Detecter
a,b,c = map(int,input().split())
if a+b>c and b+c>a and c+a>b:
    if a==b==c:
        print("Equilateral Triangle")
    elif a==b != c:
        print("Isosceles Triangle")
    else:
        print("Scalene Triangle")
else:
    print("Not a valid triangle")

12. Greatest of three numbers
a=int(input())
b=int(input())
c=int(input())
if a>b and a>c:
    print(f"{a} is the greatest")
elif b>c and b>a:
    print(f"{b} is the greatest")
elif c>a and c>b:
    print(f"{c} is the greatest")
else:
    print("There is a tie for the greatest number")
